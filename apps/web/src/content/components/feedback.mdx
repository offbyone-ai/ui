## Usage

### FeedbackButton

Drop into any nav bar — opens the form in a dialog:

```tsx
import { FeedbackButton } from "@/components/feedback-form"

export function Nav() {
  return (
    <nav>
      {/* ... */}
      <FeedbackButton appId={config.pay.appId} payUrl={config.pay.url} />
    </nav>
  )
}
```

### FeedbackForm

Use the form directly when you want to embed it in your own dialog, drawer, or page:

```tsx
import { FeedbackForm } from "@/components/feedback-form"

export function FeedbackPage() {
  return <FeedbackForm appId={config.pay.appId} payUrl={config.pay.url} />
}
```

### With a success callback

```tsx
import { FeedbackForm } from "@/components/feedback-form"

export function FeedbackDialog({ onClose }: { onClose: () => void }) {
  return (
    <FeedbackForm
      appId={config.pay.appId}
      payUrl={config.pay.url}
      onSuccess={onClose}
    />
  )
}
```

## API Reference

### FeedbackButton

| Prop | Type | Required | Description |
|------|------|----------|-------------|
| `appId` | `string` | ✓ | Unique app identifier — used to filter feedback in the dashboard |
| `payUrl` | `string` | ✓ | Base URL of your pay service (e.g. `config.pay.url`) |
| `label` | `string` | — | Button text. Renders as icon-only when omitted |
| `title` | `string` | — | Dialog title. Defaults to `"{appId} feedback"` |
| `description` | `string` | — | Optional subtitle shown below the dialog title |

### FeedbackForm

| Prop | Type | Required | Description |
|------|------|----------|-------------|
| `appId` | `string` | ✓ | Unique app identifier — used to filter feedback in the dashboard |
| `payUrl` | `string` | ✓ | Base URL of your pay service (e.g. `config.pay.url`) |
| `onSuccess` | `() => void` | — | Called after a successful submission. Useful for closing a dialog |

### Payload

The form posts JSON to `POST {payUrl}/api/feedback`:

```json
{
  "app_id": "my-app",
  "type": "bug",
  "message": "The export button does nothing on Safari.",
  "email": "alice@example.com",
  "phone": "+1 555 000 0000"
}
```

`email` is pulled automatically from `localStorage` if the user has a stored license — they never type it. `phone` is omitted when the user doesn't check "Keep me posted".
